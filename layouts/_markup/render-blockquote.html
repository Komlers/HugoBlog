{{- /* GitHub-style Markdown Alerts render hook. Hugo natively parses "> [!NOTE]" etc. and provides: .Type -> "alert" (or "regular") .AlertType -> lowercase alert type, e.g. "note", "tip", "warning" .AlertTitle -> optional custom title .Text -> blockquote content (marker line already removed) */ -}} {{- if eq .Type "alert" -}} {{- $type := .AlertType -}} {{- $class := $type | default "note" -}} {{- $title := "" -}} {{- if .AlertTitle -}} {{- $title = .AlertTitle -}} {{- else -}} {{- $titleKey := printf "ghAlert%s" $type -}} {{- $title = i18n $titleKey | default (title $type) -}} {{- end -}} {{- $svgName := "" -}} {{- if eq $type "note" }}{{ $svgName = "admonition_info" }}{{- end -}} {{- if eq $type "tip" }}{{ $svgName = "admonition_tip" }}{{- end -}} {{- if eq $type "important" }}{{ $svgName = "admonition_info" }}{{- end -}} {{- if eq $type "warning" }}{{ $svgName = "admonition_warning" }}{{- end -}} {{- if eq $type "caution" }}{{ $svgName = "admonition_danger" }}{{- end -}}
{{- partial "svg.html" (dict "context" . "name" $svgName) -}}{{ $title }}
{{ .Text | safeHTML }}
{{- else -}}
{{ .Text | safeHTML }}
{{- end -}}
